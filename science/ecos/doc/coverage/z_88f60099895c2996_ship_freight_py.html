76def period_gap(d0: float, d_hat: float, K_L: float, T: float) -> float: 

77 """ 

78 Period gap G = integral of u over one period (closed form, Lemma 1). 

79 

80 With m = d0 - K_L and c = m / d_hat: 

81 c <= -1: G = 0; |c| < 1: arcsine formula; c >= 1: G = m T. 

82 """ 

83 _check_demand(d0, d_hat, T) 

84 _nonnegative("K_L", K_L) 

85 m = d0 - K_L 

86 c = m / d_hat 

87 if c <= -1.0: 

88 return 0.0 

89 if c >= 1.0: 

90 return m * T 

91 return d_hat * T / math.pi * ( 

92 c * (math.pi / 2.0 + math.asin(c)) + math.sqrt(1.0 - c * c) 

93 )

96def _positive_intervals(m: float, d_hat: float) -> list: 

97 """Phase intervals in [0, 2 pi) where m + d_hat sin(theta) > 0.""" 

98 c = m / d_hat 

99 if c >= 1.0: 

100 return [(0.0, 2.0 * math.pi)] 

101 if c <= -1.0: 

102 return [] 

103 alpha = math.asin(c) 

104 if alpha >= 0.0: 

105 return [(0.0, math.pi + alpha), (2.0 * math.pi - alpha, 2.0 * math.pi)] 

106 return [(-alpha, math.pi + alpha)]

109def cumulative_gap(t: float, d0: float, d_hat: float, K_L: float, T: float) -> float: 

110 """Exact U(t) = integral_0^t u(s) ds for t >= 0.""" 

111 _nonnegative("t", t) 

112 G = period_gap(d0, d_hat, K_L, T) 

113 m = d0 - K_L 

114 n = math.floor(t / T) 

115 theta = 2.0 * math.pi * (t - n * T) / T 

116 partial = 0.0 

117 for a, b in _positive_intervals(m, d_hat): 

118 hi = min(b, theta) 

119 if hi > a: 

120 partial += m * (hi - a) + d_hat * (math.cos(a) - math.cos(hi)) 

121 return n * G + partial * T / (2.0 * math.pi)

198def stock_at_calls(N: int, L: float, d0: float, d_hat: float, K_L: float, T: float, 

199 I0: float, t0: float = 0.0, n_periods: int = 1) -> StockTrajectory: 

200 """ 

201 Exact stock X(t) = I0 + A(t) - U(t) at the call times t_k = t0 + k T / N 

202 (Definitions 4-5). X is non-increasing between calls and jumps by +L at 

203 each call, so suprema/infima are attained at these points. 

204 """ 

205 if N < 1: 

206 raise ValueError(f"N must be >= 1, got {N}") 

207 if n_periods < 1: 

208 raise ValueError(f"n_periods must be >= 1, got {n_periods}") 

209 _positive("L", L) 

210 _positive("T", T) 

211 if not 0.0 <= t0 < T / N: 

212 raise ValueError("t0 must be in [0, T/N)") 

213 G = period_gap(d0, d_hat, K_L, T) 

214 k = np.arange(N * n_periods) 

215 times = t0 + k * T / N 

216 U = np.array([cumulative_gap(float(t), d0, d_hat, K_L, T) for t in times]) 

217 before = I0 + k * L - U 

218 return StockTrajectory(times=times, before=before, after=before + L, G=G, L=L)

221def first_overflow_time(traj: StockTrajectory, B: float) -> Optional[float]: 

222 """t_B = inf{t : X(t) > B}; attained at a call time. None if not reached.""" 

223 hits = np.nonzero(traj.after > B)[0] 

224 if hits.size == 0: 

225 return None 

226 return float(traj.times[hits[0]])

229def first_shortage_time(traj: StockTrajectory, I0: float, d0: float, d_hat: float, 

230 K_L: float, T: float) -> Optional[float]: 

231 """t_F = inf{t : X(t) < 0}, located by root finding between calls.""" 

232 hits = np.nonzero(traj.before < 0.0)[0] 

233 if hits.size == 0: 

234 return None 

235 k = int(hits[0]) 

236 if k == 0: 

237 t_prev, x_prev = 0.0, I0 

238 else: 

239 t_prev, x_prev = float(traj.times[k - 1]), float(traj.after[k - 1]) 

240 t_next = float(traj.times[k]) 

241 U_prev = cumulative_gap(t_prev, d0, d_hat, K_L, T) 

242 

243 def x_of(t: float) -> float: 

244 return x_prev - (cumulative_gap(t, d0, d_hat, K_L, T) - U_prev) 

245 

246 return float(brentq(x_of, t_prev, t_next))

260def _relative_stock_levels(N: int, d0: float, d_hat: float, K_L: float, T: float, 

261 t0: float) -> Tuple[float, np.ndarray]: 

262 """Return L = G/N and y_k = X(t_k^-) - I0 for k = 0..N-1 (Q = G).""" 

263 G = period_gap(d0, d_hat, K_L, T) 

264 L = load_for_departures(G, N) 

265 if not 0.0 <= t0 < T / N: 

266 raise ValueError("t0 must be in [0, T/N)") 

267 edges = np.array([cumulative_gap(t0 + k * T / N, d0, d_hat, K_L, T) 

268 for k in range(N + 1)]) 

269 consumed = np.diff(edges) 

270 y = np.concatenate(([-edges[0]], -edges[0] + np.cumsum(L - consumed)))[:N] 

271 return L, y

564def _check_cells(gaps: Sequence[float], buffers: Sequence[float]) -> None: 

565 if len(gaps) == 0 or len(gaps) != len(buffers): 

566 raise ValueError("gaps and buffers must be non-empty and of equal length") 

567 for g, b in zip(gaps, buffers): 

568 _positive("cell gap", g) 

569 _positive("cell buffer", b)

572def cell_shares(gaps: Sequence[float]) -> np.ndarray: 

573 """Unloading shares s_j = G_j / G forced by the cell balance.""" 

574 arr = np.asarray(gaps, dtype=float) 

575 if arr.size == 0 or not np.all(arr > 0): 

576 raise ValueError("gaps must be non-empty and positive") 

577 return arr / arr.sum() 

578 

579 

580def cell_load_limit(gaps: Sequence[float], buffers: Sequence[float]) -> float: 

581 """L_cells = G min_j B_j / G_j: the weakest cell limits the cargo (Satz 7.2).""" 

582 _check_cells(gaps, buffers) 

583 g = np.asarray(gaps, dtype=float) 

584 b = np.asarray(buffers, dtype=float) 

585 return float(g.sum() * np.min(b / g))

587 

588def pool_load_limit(gaps: Sequence[float], buffers: Sequence[float]) -> float: 

589 """L_pool = sum_j B_j under free redistribution (Satz 7.3).""" 

590 _check_cells(gaps, buffers) 

591 return float(np.sum(buffers))

843def log_growth(omega: float, h0: float) -> float: 

844 """ 

845 g_h0(Omega) = ln(1 + (e^Omega - 1) e^-h0): log increase of the count after 

846 growth work Omega (eq. sb-lsg); g = Omega for h0 = 0. Evaluated in the

887def population_ode(mu: Callable[[float], float], t_end: float, h0: float, 

888 y0: float = 0.0, n_points: int = 201, 

889 max_step: float = 0.05) -> Tuple[np.ndarray, np.ndarray]: 

890 """ 

891 Numerical solution of the population model (eq. sb-pop), 

892 y' = mu Q/(1+Q), Q' = mu Q, Q(0) = 1/(e^h0 - 1), y(0) = y0 (h0 = 0: y' = mu). 

893 Used to check Satz sb-loesung against the closed form; returns (t, y). 

894 """ 

895 _positive("t_end", t_end) 

896 _nonnegative("h0", h0) 

897 if n_points < 2: 

898 raise ValueError(f"n_points must be >= 2, got {n_points}") 

899 t_eval = np.linspace(0.0, t_end, n_points) 

900 if h0 == 0.0: 

901 def rhs(t, state): 

902 return [mu(t)] 

903 state0 = [y0] 

904 else: 

905 def rhs(t, state): 

906 q = state[1] 

907 return [mu(t) * q / (1.0 + q), mu(t) * q] 

908 state0 = [y0, 1.0 / math.expm1(h0)] 

909 sol = solve_ivp(rhs, (0.0, t_end), state0, t_eval=t_eval, max_step=max_step, 

910 rtol=1e-9, atol=1e-12) 

911 return sol.t, sol.y[0]

1193def atmosphere_exposure(length: float, c0: float, c_inf: float, leak: float, 

1194 kappa: float) -> float: 

1195 """

1197 with A = 1 + kappa c_inf, B = kappa (c0 - c_inf) (eq. sb-X). Then 

1198 Omega = mu_0 X(l) at constant product temperature. 

1199 """ 

1200 _nonnegative("length", length) 

1201 _nonnegative("c0", c0) 

1202 _nonnegative("c_inf", c_inf) 

1203 _positive("leak", leak) 

1204 _nonnegative("kappa", kappa)

1206 b = kappa * (c0 - c_inf) 

1207 # ln(A e^{kl} + B) = kl + ln(A + B e^{-kl}): no overflow for long voyages

1355def critical_voyage_duration(params: ShipMarketParams, mu_sea: float, mu_store: float, 

1356 omega_s: float, omega_fixed: float, tau0_days: float = 0.0, 

1357 y: float = 0.0, sigma_omega: float = 0.0, 

1358 days_per_horizon_unit: float = 365.0) -> Optional[float]: 

1359 """ 

1360 Critical voyage duration tau_S^krit of the double clamp (Satz sb-klemme): 

1361 the largest tau_S (days) with L_eps(tau)^- <= L_F(tau_S), where 

1362 tau = tau0 + tau_S (planning horizon), 

1363 Omega_vor + Omega_nach = omega_fixed + mu_S tau_S 

1364 (omega_fixed = all work that does 

1365 not scale with the voyage), 

1366 L_F = u_bar a_ok - y, u_bar = G / T (kt per day; T converted to days), 

1367 and eps(tau) from the degradation function of `params`; L_eps^- uses the 

1368 economic parameters of `params`. `days_per_horizon_unit` converts days to 

1369 the time unit of tau (the economic chapter uses years, hence 365). 

1370 Returns None if no duration is feasible (already tau_S = 0 fails, or no 

1371 load is viable). With sigma_omega = 0 no uncertainty discount is applied. 

1372 """ 

1373 _positive("mu_store", mu_store) 

1374 _nonnegative("mu_sea", mu_sea) 

1375 _positive("omega_s", omega_s) 

1376 _nonnegative("omega_fixed", omega_fixed) 

1377 _nonnegative("tau0_days", tau0_days) 

1378 _nonnegative("y", y) 

1379 _nonnegative("sigma_omega", sigma_omega) 

1380 _positive("days_per_horizon_unit", days_per_horizon_unit) 

1381 P = params 

1382 u_bar = P.G / (P.T * days_per_horizon_unit) # kt per day (T is in horizon units) 

1383 

1384 def gap_to_economics(tau_s: float) -> float: 

1385 horizon = (tau0_days + tau_s) / days_per_horizon_unit 

1386 delta = degradation_exponential(horizon, P.delta_max, P.lam) 

1387 eps = safety_margin(P.sigma_m, delta, P.p) 

1388 l_minus = just_worth_load(P.a0, eps, P.Cf, P.kappa) 

1389 if l_minus is None: 

1390 return -math.inf 

1391 budget = omega_s 

1392 if sigma_omega > 0.0: 

1393 budget = effective_budget(omega_s, sigma_omega, delta, P.p) 

1394 a_ok = admissible_storage_age(budget, omega_fixed + mu_sea * tau_s, 0.0, 

1395 mu_store) 

1396 return freshness_load_limit(u_bar, a_ok, y) - l_minus 

1397 

1398 if gap_to_economics(0.0) < 0.0: 

1399 return None 

1400 hi = 1.0 

1401 while gap_to_economics(hi) >= 0.0: 

1402 hi *= 2.0 

1403 lo = 0.0 

1404 for _ in range(200): 

1405 mid = 0.5 * (lo + hi) 

1406 if gap_to_economics(mid) >= 0.0: 

1407 lo = mid 

1408 else: 

1409 hi = mid 

1410 return lo
