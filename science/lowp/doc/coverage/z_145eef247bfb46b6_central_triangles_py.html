1"""Acht Zentraldreiecke mit Spitze im Mittelpunkt: Rechenmodul zum Kapitel 

2"Acht Zentraldreiecke mit Spitze im Mittelpunkt" (lowp.tex, Label sec:ok). 

3 

4Nur Standardbibliothek. Die Funktionen sind nach den Sätzen des Kapitels benannt; 

5der Standardwert ``n = 8`` ist die Achtelung (drei Teilungen), alle Funktionen 

6gelten aber für jedes reguläre n-Eck mit n >= 3, soweit der Satz das hergibt. 

7 

8Notation (wie im Kapitel) 

9 K Einheitskreis, Mittelpunkt M = 0 

10 omega = exp(i*pi/4) (allgemein exp(2*pi*i/n)) 

11 P_k = omega^k (Ecken, Indizes modulo n) 

12 T_k = conv{0, P_k, P_(k+1)} (Zentraldreieck) 

13 A_n, B_n = Flächen des ein- bzw. umbeschriebenen regulären n-Ecks 

14 t_n = tan(pi/n) (Halbbasis / Höhe) 

15"""

175def triangle_coordinates(z: complex, k: int, n: int = 8) -> tuple[float, float]: 

176 """Eindeutige Koordinaten (a, b) mit z = a*P_k + b*P_(k+1) (Lemma Sektorkegel).""" 

177 _check_n(n) 

178 pk, pk1 = vertex(k, n), vertex(k + 1, n) 

179 sin_alpha = math.sin(2.0 * math.pi / n) 

180 b = (z * pk.conjugate()).imag / sin_alpha 

181 a = (pk1.conjugate() * z).imag * -1.0 / sin_alpha 

182 return a, b

243def central_angles(args: Sequence[float]) -> list[float]: 

244 """Winkelabstände alpha_j = phi_(j+1) - phi_j der streng wachsenden Argumente (zyklisch). 

245 

246 Voraussetzung: mindestens 3 Punkte, Argumente in [0, 2*pi), streng wachsend. 

247 """ 

248 phis = list(args) 

249 if len(phis) < 3: 

250 raise ValueError("mindestens drei Punkte erforderlich") 

251 if any(not 0.0 <= p < 2.0 * math.pi for p in phis): 

252 raise ValueError("Argumente müssen in [0, 2*pi) liegen") 

253 if any(b <= a for a, b in zip(phis, phis[1:])): 

254 raise ValueError("Argumente müssen streng wachsen") 

255 gaps = [b - a for a, b in zip(phis, phis[1:])] 

256 gaps.append(phis[0] + 2.0 * math.pi - phis[-1]) 

257 return gaps

336def doubled_areas(a_n: float, b_n: float) -> tuple[float, float]: 

337 """Verdopplung n -> 2n aus (A_n, B_n): A_2n = sqrt(A_n*B_n), B_2n = 2*A_2n*B_n/(A_2n+B_n).""" 

338 if a_n <= 0.0 or b_n <= 0.0: 

339 raise ValueError("Flächen müssen positiv sein")

345def refine_bounds(m_start: int = 2, steps: int = 5) -> list[tuple[int, float, float]]: 

346 """Einschließungen für n = 2^(m_start), 2^(m_start+1), ... allein über die Mittelwertformeln. 

347 

348 Start mit dem Quadrat (m_start = 2): A_4 = 2, B_4 = 4. Liefert (n, A_n, B_n). 

349 """ 

350 if _check_nonneg(m_start, "m_start") < 2: 

351 raise ValueError("m_start muss >= 2 sein (n >= 4)") 

352 _check_nonneg(steps, "steps") 

353 n = 2 ** m_start 

354 a, b = inscribed_area(n), circumscribed_area(n) 

355 rows = [(n, a, b)] 

356 for _ in range(steps): 

357 a, b = doubled_areas(a, b) 

358 n *= 2 

359 rows.append((n, a, b)) 

360 return rows

417def arctan_partial(t: float, n_terms: int) -> float: 

418 """Partialsumme sum_{j<N} (-1)^j t^(2j+1)/(2j+1) der Arkustangensreihe, t in [0, 1].""" 

419 if not 0.0 <= t <= 1.0: 

420 raise ValueError("t muss in [0, 1] liegen") 

421 if isinstance(n_terms, bool) or not isinstance(n_terms, int) or n_terms < 1: 

422 raise ValueError("n_terms muss eine ganze Zahl >= 1 sein") 

423 return math.fsum((-1) ** j * t ** (2 * j + 1) / (2 * j + 1) for j in range(n_terms)) 

424 

425 

426def arctan_remainder_bound(t: float, n_terms: int) -> float: 

427 """Restgliedschranke t^(2N+1)/(2N+1) (Lemma Arkustangens).""" 

428 if not 0.0 <= t <= 1.0: 

429 raise ValueError("t muss in [0, 1] liegen") 

430 if isinstance(n_terms, bool) or not isinstance(n_terms, int) or n_terms < 1: 

431 raise ValueError("n_terms muss eine ganze Zahl >= 1 sein") 

432 return t ** (2 * n_terms + 1) / (2 * n_terms + 1)

435def central_triangle_series(n: int, n_terms: int) -> float: 

436 """Partialsumme S_N^(n) = n * sum_{j<N} (-1)^j t_n^(2j+1)/(2j+1) (Satz Zentraldreiecksreihe).""" 

437 _check_n(n) 

438 if n < 4:

471 Links sum (-1)^j/(2j+1), rechts 2*sum (-1)^j (sqrt2-1)^(2j+1)/(2j+1); beide -> pi/4. 

472 """ 

473 left = arctan_partial(1.0, n_terms) 

474 right = 2.0 * arctan_partial(math.sqrt(2.0) - 1.0, n_terms) 

475 return left, right

478def double_angle_identity_holds(tol: float = 1e-12) -> bool: 

479 """Algebraische Probe tan(2x) = 2*tan(x)/(1 - tan(x)^2) = 1 für x = arctan(sqrt 2 - 1).""" 

480 t = math.sqrt(2.0) - 1.0 

481 return abs(2.0 * t / (1.0 - t * t) - 1.0) <= tol

485# Satz Halbierung des Zentraldreiecks, Beobachtung Gliederzahl 

486# --------------------------------------------------------------------------

488def tau_exact(k: int) -> float: 

489 """tau_k = tan(pi / 2^(k+2)) direkt (tau_0 = 1, tau_1 = sqrt 2 - 1).""" 

490 if _check_nonneg(k, "k") == 0: 

491 return 1.0 # tan(pi/4) = 1 exakt (Gleitkomma liefert 0,9999999999999999) 

492 return math.tan(math.pi / 2 ** (k + 2)) 

493 

494 

495def tau_next(tau: float) -> float: 

496 """Halbierungsrekursion tau_(k+1) = tau_k / (1 + sqrt(1 + tau_k^2)).""" 

497 if tau < 0.0: 

498 raise ValueError("tau muss >= 0 sein") 

499 return tau / (1.0 + math.sqrt(1.0 + tau * tau)) 

500 

501 

502def tau_sequence(k_max: int) -> list[float]: 

503 """tau_0, ..., tau_k_max über die Rekursion, beginnend bei tau_0 = 1.""" 

504 seq = [1.0] 

505 for _ in range(_check_nonneg(k_max, "k_max")): 

506 seq.append(tau_next(seq[-1])) 

507 return seq 

508 

509 

510def tau_degree_bound(k: int) -> int: 

511 """Grad-Schranke 2^k: tau_k ist algebraisch vom Grad <= 2^k über Q.""" 

512 return 2 ** _check_nonneg(k, "k") 

513 

514 

515def pi_from_tau(k: int, n_terms: int) -> float: 

516 """pi = 2^(k+2) * sum (-1)^j tau_k^(2j+1)/(2j+1), Partialsumme mit N Gliedern.""" 

517 return 2 ** (_check_nonneg(k, "k") + 2) * arctan_partial(tau_exact(k), n_terms) 

518 

519 

520def terms_needed(k: int, eps: float) -> int: 

521 """Kleinstes N mit 2^(k+2) * tau_k^(2N+1)/(2N+1) < eps (garantierte Gliederzahl).""" 

522 if not eps > 0.0: 

523 raise ValueError("eps muss positiv sein") 

524 n_corners = 2 ** (_check_nonneg(k, "k") + 2) 

525 tau = tau_exact(k) 

526 

527 def bound(n_terms: int) -> float: 

528 return n_corners * tau ** (2 * n_terms + 1) / (2 * n_terms + 1) 

529 

530 hi = 1 

531 while bound(hi) >= eps: 

532 hi *= 2 

533 lo = hi // 2 if hi > 1 else 0 # bound(lo) >= eps (oder lo = 0), bound(hi) < eps 

534 while hi - lo > 1: 

535 mid = (lo + hi) // 2 

536 if bound(mid) < eps: 

537 hi = mid 

538 else: 

539 lo = mid 

540 return hi 

541 

542 

543def terms_needed_table(ks: Iterable[int] = range(0, 6), 

544 eps_values: Sequence[float] = (1e-6, 1e-10)) -> list[dict[str, float]]: 

545 """Tabelle der Beobachtung 'Wie viele Glieder braucht die Teilung?'.""" 

546 rows = [] 

547 for k in ks: 

548 row: dict[str, float] = { 

549 "k": k, 

550 "corners": 2 ** (k + 2), 

551 "tau": tau_exact(k), 

552 } 

553 for eps in eps_values: 

554 row[f"N_{eps:g}"] = terms_needed(k, eps) 

555 rows.append(row) 

556 return rows

563def mercator_partial(theta: float, n_terms: int) -> complex: 

564 """Partialsumme sum_{n=1}^N (-1)^(n+1) exp(i*n*theta)/n.""" 

565 if isinstance(n_terms, bool) or not isinstance(n_terms, int) or n_terms < 1: 

566 raise ValueError("n_terms muss eine ganze Zahl >= 1 sein") 

567 return sum((-1) ** (n + 1) * cmath.exp(1j * n * theta) / n for n in range(1, n_terms + 1))

570def mercator_limit(theta: float) -> complex: 

571 """Grenzwert ln(2*cos(theta/2)) + i*theta/2 für theta in (-pi, pi).""" 

572 if not -math.pi < theta < math.pi: 

573 raise ValueError("theta muss in (-pi, pi) liegen") 

574 return complex(math.log(2.0 * math.cos(theta / 2.0)), theta / 2.0)

577def eighth_sine_limit() -> float: 

578 """sum (-1)^(n+1) sin(n*pi/4)/n = pi/8.""" 

579 return math.pi / 8.0

581 

582def eighth_cosine_limit() -> float: 

583 """sum (-1)^(n+1) cos(n*pi/4)/n = (1/2)*ln(2 + sqrt 2).""" 

584 return 0.5 * math.log(2.0 + math.sqrt(2.0))

586 

587def newton_signs(kind: int) -> tuple[int, int, int, int]: 

588 """Vorzeichenmuster der Achtel-Reihen, Periode 4 in j. 

589 

590 kind 1: (+, +, -, -) -> Vorzeichen von Im(omega^(2j+1)), Reihe mit Wert pi/(2*sqrt 2) 

591 kind 2: (+, -, -, +) -> Vorzeichen von Re(omega^(2j+1)), Reihe mit Wert ln(1+sqrt 2)/sqrt 2 

592 """ 

593 if kind == 1: 

594 return (1, 1, -1, -1) 

595 if kind == 2: 

596 return (1, -1, -1, 1) 

597 raise ValueError("kind muss 1 oder 2 sein") 

598 

599 

600def newton_series_partial(kind: int, n_terms: int) -> float: 

601 """Partialsumme sum_{j<N} eps_j/(2j+1) mit dem Vorzeichenmuster ``kind``.""" 

602 if isinstance(n_terms, bool) or not isinstance(n_terms, int) or n_terms < 1: 

603 raise ValueError("n_terms muss eine ganze Zahl >= 1 sein") 

604 signs = newton_signs(kind) 

605 return math.fsum(signs[j % 4] / (2 * j + 1) for j in range(n_terms)) 

606 

607 

608def newton_series_limit(kind: int) -> float: 

609 """Grenzwerte pi/(2*sqrt 2) (kind 1) bzw. ln(1 + sqrt 2)/sqrt 2 (kind 2).""" 

610 if kind == 1: 

611 return math.pi / (2.0 * math.sqrt(2.0)) 

612 if kind == 2: 

613 return math.log(1.0 + math.sqrt(2.0)) / math.sqrt(2.0) 

614 raise ValueError("kind muss 1 oder 2 sein") 

615 

616 

617def newton_series_extrapolated(kind: int, n_terms: int) -> float: 

618 """Richardson-Extrapolation 2*S_(2N) - S_N der Partialsummen (N = n_terms Glieder). 

619 

620 Die Partialsummen S_N der Achtel-Reihen haben einen Fehler c/N + O(1/N^2); die 

621 Kombination hebt den 1/N-Term auf und konvergiert deutlich schneller. 

622 """ 

623 return 2.0 * newton_series_partial(kind, 2 * n_terms) - newton_series_partial(kind, n_terms)

630def chord(k: int, n: int = 8) -> float:

643def chord_values_8() -> dict[int, float]:

673 

674def dihedral_group(n: int = 8) -> list[tuple[int, bool]]: 

675 """Elemente (m, spiegelung) der Diedergruppe D_n: z -> w^m z bzw. z -> w^m conj(z).""" 

676 _check_n(n) 

677 return [(m, refl) for refl in (False, True) for m in range(n)] 

678 

679 

680def apply_symmetry(g: tuple[int, bool], z: complex, n: int = 8) -> complex: 

681 """Wendet ein Gruppenelement (m, spiegelung) auf z an.""" 

682 m, refl = g 

683 w = omega(n) ** m 

684 return w * (z.conjugate() if refl else z) 

685 

686 

687def group_order(n: int = 8) -> int: 

688 """|D_n| = 2n (n = 8: 16).""" 

689 return 2 * _check_n(n) 

690 

691 

692def maps_vertices_to_vertices(g: tuple[int, bool], n: int = 8, tol: float = 1e-9) -> bool: 

693 """Permutiert g die Eckenmenge {P_k}? (Bedingung für g(O) = O).""" 

694 verts = vertices(n) 

695 return all(any(abs(apply_symmetry(g, p, n) - v) <= tol for v in verts) for p in verts) 

696 

697 

698def triangle_image(g: tuple[int, bool], k: int, n: int = 8) -> int: 

699 """Index j mit g(T_k) = T_j für ein Element g der Diedergruppe D_n. 

700 

701 Elemente von D_n bilden benachbarte Ecken auf benachbarte Ecken ab; das Bild von T_k ist 

702 daher wieder ein Zentraldreieck. Drehungen erhalten, Spiegelungen vertauschen die Reihenfolge 

703 der beiden Basisecken. 

704 """ 

705 _check_n(n) 

706 i1 = _vertex_index(apply_symmetry(g, vertex(k, n), n), n) 

707 i2 = _vertex_index(apply_symmetry(g, vertex(k + 1, n), n), n) 

708 return i1 if i2 == (i1 + 1) % n else i2 

709 

710 

711def _vertex_index(z: complex, n: int) -> int: 

712 """Index k mit P_k = z (z muss eine Ecke sein).""" 

713 k = round(cmath.phase(z) / (2.0 * math.pi / n)) % n 

714 return int(k)

717def orbit_of_triangle(k: int = 0, n: int = 8) -> set[int]: 

718 """Bahn von T_k unter D_n; transitiv, also alle n Dreiecke.""" 

719 return {triangle_image(g, k, n) for g in dihedral_group(n)}

721 

722def stabilizer(k: int = 0, n: int = 8) -> list[tuple[int, bool]]: 

723 """Stabilisator von T_k in D_n (Ordnung 2: Identität und Spiegelung an der Höhe).""" 

724 return [g for g in dihedral_group(n) if triangle_image(g, k, n) == k % n]

726 

727def is_isometry_of_polygon(g: Callable[[complex], complex], n: int = 8, tol: float = 1e-9) -> bool: 

728 """Prüft an Stichproben, ob eine Abbildung O auf sich abbildet und Abstände erhält.""" 

729 verts = vertices(n) 

730 for a in verts: 

731 for b in verts: 

732 if abs(abs(g(a) - g(b)) - abs(a - b)) > tol: 

733 return False 

734 return all(any(abs(g(a) - v) <= tol for v in verts) for a in verts)

746def viete_factors(m: int) -> list[float]: 

747 """Faktoren cos(pi/2^k), k = 2..m, über die Radikalrekursion a_(j+1) = sqrt(2 + a_j).""" 

748 if isinstance(m, bool) or not isinstance(m, int) or m < 2: 

749 raise ValueError("m muss eine ganze Zahl >= 2 sein") 

750 factors = [] 

751 a = math.sqrt(2.0) # 2*cos(pi/4) 

752 for _ in range(2, m + 1): 

753 factors.append(a / 2.0) 

754 a = math.sqrt(2.0 + a) 

755 return factors

759 """prod_{k=2}^m cos(pi/2^k).""" 

760 return math.prod(viete_factors(m))

764 """A_(2^(m+1)) = A_4 / prod_{k=2}^m cos(pi/2^k) mit A_4 = 2.""" 

765 return 2.0 / viete_product(m)

767 

768def viete_pi(m: int) -> float: 

769 """Näherung 2 / prod_{k=2}^m cos(pi/2^k) -> pi.""" 

770 return viete_area(m)
