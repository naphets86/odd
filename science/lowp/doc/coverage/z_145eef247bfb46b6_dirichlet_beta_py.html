2Dirichlet Beta-Funktion und Catalan-Konstante

29 Initialisierung der Dirichlet Beta-Funktion.

34 self.tolerance = tolerance

37 def dirichlet_beta_series(self, s: complex, n_terms: int = 1000) -> complex: 

38 """ 

39 Berechnung von β(s) über die definierende Reihe für Re(s) > 0. 

40  

41 β(s) = Σ((-1)^n / (2n+1)^s) für n=0 bis ∞ 

42  

43 Args: 

44 s: Komplexer Parameter 

45 n_terms: Anzahl der Reihenglieder 

46  

47 Returns: 

48 Näherung von β(s) 

49  

50 Raises: 

51 ValueError: Wenn Re(s) ≤ 0 

52 """ 

53 if np.real(s) <= 0: 

54 raise ValueError(f"Reihe konvergiert nur für Re(s) > 0, erhalten: Re(s) = {np.real(s)}") 

55 

56 result = 0.0j if isinstance(s, complex) else 0.0 

57 for n in range(n_terms): 

58 term = ((-1)**n) / ((2*n + 1)**s) 

59 result += term 

60 

61 # Abbruch bei Konvergenz 

62 if abs(term) < self.tolerance: 

63 break 

64 

65 return result

67 def catalan_constant_series(self, n_terms: int = 10000) -> float: 

68 """ 

69 Berechnung der Catalan-Konstante G = β(2). 

70  

71 G = Σ((-1)^n / (2n+1)^2) = 1 - 1/9 + 1/25 - 1/49 + ... 

72  

73 Args: 

74 n_terms: Anzahl der Reihenglieder 

75  

76 Returns: 

77 Näherung der Catalan-Konstante 

78 """ 

79 result = 0.0 

80 for n in range(n_terms): 

81 term = ((-1)**n) / ((2*n + 1)**2) 

82 result += term 

83 

84 if abs(term) < self.tolerance: 

85 break 

86 

87 return result

89 def catalan_arctan_integral(self, n_quad: int = 500) -> float: 

90 """ 

91 Berechnung der Catalan-Konstante via Arcustangens-Integral. 

92  

93 G = ∫₀¹ arctan(x)/x dx 

94  

95 Args: 

96 n_quad: Anzahl der Quadraturpunkte 

97  

98 Returns: 

99 Näherung der Catalan-Konstante 

100 """ 

101 def integrand(x): 

102 if x == 0: 

103 return 1.0 # lim_{x→0} arctan(x)/x = 1 

104 return np.arctan(x) / x 

105 

106 result, _ = quad(integrand, 0, 1, limit=n_quad) 

107 return result

109 def catalan_logarithmic_integral(self, n_quad: int = 500) -> float: 

110 """ 

111 Berechnung der Catalan-Konstante via logarithmisches Integral. 

112  

113 G = -∫₀¹ ln(x)/(1+x²) dx 

114  

115 (Hinweis: -∫₀¹ ln(1+x²)/(2x) dx = -π²/48 ist NICHT G.) 

116  

117 Args: 

118 n_quad: Anzahl der Quadraturpunkte 

119  

120 Returns: 

121 Näherung der Catalan-Konstante 

122 """ 

123 def integrand(x): 

124 if x == 0: 

125 return 0.0 # Randpunkt (von quad nicht ausgewertet) 

126 return -np.log(x) / (1 + x**2) 

127 

128 result, _ = quad(integrand, 0, 1, limit=n_quad) 

129 return result

131 def catalan_sine_integral(self, n_quad: int = 500) -> float: 

132 """ 

133 Berechnung der Catalan-Konstante via Sinus-Integral. 

134  

135 G = ½ ∫₀^(π/2) x/sin(x) dx 

136  

137 (Hinweis: ∫₀^(π/2) x/sin(x) dx = 2G, der Faktor ½ ist nötig.) 

138  

139 Args: 

140 n_quad: Anzahl der Quadraturpunkte 

141  

142 Returns: 

143 Näherung der Catalan-Konstante 

144 """ 

145 def integrand(x): 

146 if x == 0: 

147 return 1.0 # lim_{x→0} x/sin(x) = 1 

148 return x / np.sin(x) 

149 

150 result, _ = quad(integrand, 0, np.pi/2, limit=n_quad) 

151 return 0.5 * result

153 def beta_functional_equation(self, s: complex) -> complex: 

154 """ 

155 Anwendung der funktionalen Gleichung der Dirichlet Beta-Funktion. 

156  

157 β(s) = (2/π)^(1-s) cos(πs/2) Γ(1-s) β(1-s) 

158  

159 (Die Form 2^s π^(s-1) sin(πs/2) Γ(1-s) ζ(1-s) gilt für die 

160 Riemann-Zeta-Funktion, nicht für β.) 

161  

162 Dies erlaubt analytische Fortsetzung außerhalb Re(s) > 0. 

163 An s = 1, 2, 3, ... ist die Formel numerisch singulär (0·∞). 

164  

165 Args: 

166 s: Komplexer Parameter 

167  

168 Returns: 

169 β(s) via funktionale Gleichung 

170 """ 

171 factor1 = (2.0 / np.pi)**(1 - s) 

172 factor2 = np.cos(np.pi * s / 2) 

173 factor3 = special.gamma(1 - s) 

174 

175 # β(1-s) über die Reihe, falls Re(1-s) > 0, sonst über Fortsetzung 

176 if np.real(1 - s) > 0: 

177 beta_1_minus_s = self.dirichlet_beta_series(1 - s) 

178 else: 

179 beta_1_minus_s = self._beta_analytic_continuation(1 - s) 

180 

181 return factor1 * factor2 * factor3 * beta_1_minus_s 

182 

183 def _beta_analytic_continuation(self, s: complex) -> complex: 

184 """ 

185 Analytische Fortsetzung: Reihe für Re(s) > 0, sonst funktionale Gleichung. 

186  

187 Args: 

188 s: Komplexer Parameter 

189  

190 Returns: 

191 β(s) 

192 """ 

193 if np.real(s) > 0: 

194 return self.dirichlet_beta_series(s) 

195 # Re(s) ≤ 0 => Re(1-s) ≥ 1 > 0, die Gleichung terminiert 

196 return self.beta_functional_equation(s)

198 def poles_and_residues(self) -> dict: 

199 """ 

200 Pole und Residuen der Dirichlet Beta-Funktion. 

201  

202 β ist eine GANZE Funktion, sie hat keine Pole. Die Punkte 

203 s = -(2k+1) sind die trivialen Nullstellen (β(-k) = E_k/2 mit 

204 Euler-Zahlen E_k, E_ungerade = 0). Das Residuum ist dort 0. 

205  

206 Returns: 

207 Dictionary {s = -(2k+1): Residuum = 0.0} für die ersten 10 Punkte 

208 """ 

209 return {-(2*k + 1): 0.0 for k in range(10)}

211 def zeta_eta_relationship(self, s: complex, n_terms: int = 1000) -> Tuple[complex, complex]: 

212 """ 

213 ζ(s) (Partialsumme) und β(s) (Reihe) für Re(s) > 0. 

214  

215 β lässt sich NICHT allein aus ζ und η darstellen: die frühere Formel 

216 β = (ζ+η)/2^(s+1) ist falsch. Exakt gilt über Hurwitz-Zeta: 

217 β(s) = 4^(-s) [ζ(s,1/4) - ζ(s,3/4)]. 

218  

219 Args: 

220 s: Komplexer Parameter 

221 n_terms: Anzahl Reihenglieder 

222  

223 Returns: 

224 (β(s), ζ(s)) 

225 """ 

226 if np.real(s) <= 0: 

227 raise ValueError("Diese Methode erfordert Re(s) > 0") 

228 

229 # Riemann Zeta-Funktion (Partialsumme) 

230 zeta_s = 0.0j if isinstance(s, complex) else 0.0 

231 for n in range(1, n_terms): 

232 zeta_s += 1.0 / (n**s) 

233 

234 # Dirichlet Beta direkt aus der definierenden Reihe 

235 beta_s = self.dirichlet_beta_series(s, n_terms) 

236 

237 return beta_s, zeta_s

259 def error_estimate(self, s: float, n_terms: int) -> float: 

260 """ 

261 Fehlerschätzung für Reihenabbruch bei Re(s) > 0. 

262  

263 Der Fehler ist höchstens das erste nicht berechnete Glied. 

264 |Fehler| ≤ 1/(2n+1)^s für das n-te Glied 

265  

266 Args: 

267 s: Reeller Parameter (Re(s) > 0) 

268 n_terms: Anzahl der berechneten Glieder 

269  

270 Returns: 

271 Obere Schranke des Fehlers 

272 """ 

273 if s <= 0: 

274 raise ValueError("Erfordert s > 0") 

275 

276 # Das erste nicht berechnete Glied (n-ter Glied) 

277 return 1.0 / ((2*n_terms + 1)**s)
